cmake_minimum_required(VERSION 3.16)
project(rapidrvr C)
# Rapid River (Namco GORgON, "System 22.5") on the SHARED System 23 host (../s23/host.cmake) + its GORgON board (../s23/src/s23_gorgon.c):
# this directory adds the game's table (src/rr_game.c), its lifted program (gen/rapidrvr_lifted_NN.c: tools/import.sh, then tools/regen.sh) and its two H8 programs.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-fno-strict-aliasing -fwrapv)
include(${CMAKE_CURRENT_SOURCE_DIR}/../s23/host.cmake)
file(GLOB RR_LIFTED CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/gen/rapidrvr_lifted_*.c)
if(NOT RR_LIFTED)
  message(FATAL_ERROR "gen/rapidrvr_lifted_NN.c missing: run the lift (see the comment above)")
endif()
set_source_files_properties(${RR_LIFTED} PROPERTIES COMPILE_OPTIONS "-O1;-w;-include;${S23_HOST_DIR}/include/s23_mips_rt.h")
# THE TWO H8 PROGRAMS, translated at build time with tc2's translator from sub.bin / io.bin (tools/make_prog.py) and MAME's coverage of them
file(GLOB H8SUB_COV CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cov/h8sub_*.pcs)
file(GLOB H8IO_COV CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cov/h8io_*.pcs)
string(REPLACE ".pcs" "" H8SUB_PREFIX "${H8SUB_COV}")
string(REPLACE ".pcs" "" H8IO_PREFIX "${H8IO_COV}")
set(H8T ${S23_HOST_DIR}/tools/h8)
set(H8_TOOLS ${H8T}/h8_translate.py ${H8T}/h8_decode.py ${H8T}/h8_ops.txt ${H8T}/h8_timing.txt)
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/rr_sub.c
    COMMAND ${CMAKE_COMMAND} -E env H8_ROMSIZE=0x8000 python3 ${H8T}/h8_translate.py extracted/sub.bin 1 s23sub gen/rr_sub.c ${H8SUB_PREFIX}   # code ends ~0x7400, sound data from 0x8000 (uncapped, the table guess walked the fill: 45 MB of C)
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/extracted/sub.bin ${H8_TOOLS} ${H8SUB_COV} WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/rr_io.c
    COMMAND ${CMAKE_COMMAND} -E env H8_ROMSIZE=0x4000 python3 ${H8T}/h8_translate.py extracted/io.bin 0 s23io gen/rr_io.c ${H8IO_PREFIX}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/extracted/io.bin ${H8_TOOLS} ${H8IO_COV} WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
set(RR_SRC ${S23_HOST_SRC} src/rr_game.c ${S23_HOST_DIR}/src/s23_gorgon.c gen/rr_sub.c gen/rr_io.c ${RR_LIFTED} ${S23_HOST_TAIL})
option(PORTABLE "Bind glibc symbols to baseline versions (runs on glibc >= 2.34)" ON)
if(PORTABLE AND NOT WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_compile_options(-include ${S23_HOST_DIR}/include/glibc_compat.h)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
if(WIN32)
  set(ZLIB_USE_STATIC_LIBS ON)
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
if(WIN32)
  # Windows (MinGW-w64, ./build-windows.sh): Time Crisis 2's block -- no console, one static .exe, opengl32, Winsock for the recoil output
  add_compile_options(-include ${S23_HOST_DIR}/include/win_compat.h)
  set(RR_TARGETS rapidrvr)
  set(RR_SYS_LIBS opengl32 ws2_32)
  set(RR_LINK_OPTS -mwindows -static -static-libgcc -Wl,--stack,268435456)   # 256 MB main stack: the lifted code nests one C frame per MIPS call, and the GL driver runs on top of it
else()
  set(OpenGL_GL_PREFERENCE GLVND)
  find_package(OpenGL REQUIRED)
  set(RR_TARGETS rapidrvr rapidrvr_trace)
  set(RR_SYS_LIBS OpenGL::GL m)
  set(RR_LINK_OPTS "")
endif()
foreach(t ${RR_TARGETS})
  add_executable(${t} ${RR_SRC})
  # include/ first: its tc2_lifted.h names THIS game's lifted program
  target_include_directories(${t} PRIVATE include gen ${S23_HOST_INCLUDE} ${SDL2_INCLUDE_DIRS})
  if(WIN32)
    target_link_libraries(${t} ${SDL2_STATIC_LDFLAGS} ZLIB::ZLIB Threads::Threads ${RR_SYS_LIBS})
  else()
    target_link_libraries(${t} ${SDL2_LIBRARIES} ZLIB::ZLIB Threads::Threads ${RR_SYS_LIBS})
  endif()
  target_link_options(${t} PRIVATE ${RR_LINK_OPTS})
endforeach()
if(TARGET rapidrvr_trace)
  target_compile_definitions(rapidrvr_trace PRIVATE RR_TRACE)
endif()
