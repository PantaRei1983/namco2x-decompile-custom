cmake_minimum_required(VERSION 3.16)
project(gunwars C)
# Gunmen Wars (Namco Super System 23 + GMEN board) on the SHARED System 23 host (../s23/host.cmake): this directory adds only the game's
# table (src/gw_game.c, src/gw_gmen.c), its lifted main program (gen/gunwars_lifted_NN.c: tools/import.sh, then tools/regen.sh) and its two H8 programs.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-fno-strict-aliasing -fwrapv)
include(${CMAKE_CURRENT_SOURCE_DIR}/../s23/host.cmake)
file(GLOB GW_LIFTED CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/gen/gunwars_lifted_*.c)
if(NOT GW_LIFTED)
  message(FATAL_ERROR "gen/gunwars_lifted_NN.c missing: run the lift (see the comment above)")
endif()
set_source_files_properties(${GW_LIFTED} PROPERTIES COMPILE_OPTIONS "-O1;-w;-include;${S23_HOST_DIR}/include/s23_mips_rt.h")
# THE TWO H8 PROGRAMS, translated at build time with tc2's translator from sub.bin / io.bin (tools/make_prog.py) and MAME's coverage of them
file(GLOB H8SUB_COV CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cov/h8sub_*.pcs)
file(GLOB H8IO_COV CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cov/h8io_*.pcs)
string(REPLACE ".pcs" "" H8SUB_PREFIX "${H8SUB_COV}")
string(REPLACE ".pcs" "" H8IO_PREFIX "${H8IO_COV}")
set(H8T ${S23_HOST_DIR}/tools/h8)
set(H8_TOOLS ${H8T}/h8_translate.py ${H8T}/h8_decode.py ${H8T}/h8_ops.txt ${H8T}/h8_timing.txt)
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/gw_sub.c
    COMMAND python3 ${H8T}/h8_translate.py extracted/sub.bin 1 s23sub gen/gw_sub.c ${H8SUB_PREFIX}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/extracted/sub.bin ${H8_TOOLS} ${H8SUB_COV} WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_custom_command(OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/gen/gw_io.c
    COMMAND ${CMAKE_COMMAND} -E env H8_ROMSIZE=0x4000 python3 ${H8T}/h8_translate.py extracted/io.bin 0 s23io gen/gw_io.c ${H8IO_PREFIX}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/extracted/io.bin ${H8_TOOLS} ${H8IO_COV} WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
set(GW_SRC ${S23_HOST_SRC} src/gw_game.c src/gw_gmen.c gen/gw_sub.c gen/gw_io.c ${GW_LIFTED} ${S23_HOST_TAIL} src/gw_link.c src/gw_cam.c)   # (the link and the webcam: sockets, threads -- outside the saved state)
option(PORTABLE "Bind glibc symbols to baseline versions (runs on glibc >= 2.34)" ON)
if(PORTABLE AND NOT WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_compile_options(-include ${S23_HOST_DIR}/include/glibc_compat.h)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
if(WIN32)
  set(ZLIB_USE_STATIC_LIBS ON)
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
if(WIN32)
  # Windows (MinGW-w64, ./build-windows.sh): Time Crisis 2's block -- no console, one static .exe, opengl32, Winsock for the recoil output
  add_compile_options(-include ${S23_HOST_DIR}/include/win_compat.h)
  set(GW_TARGETS gunwars)
  set(GW_SYS_LIBS opengl32 ws2_32 mfplat mf mfreadwrite mfuuid ole32 windowscodecs)   # + Media Foundation (the webcam) and WIC (the photo file): src/gw_cam.c
  set(GW_LINK_OPTS -mwindows -static -static-libgcc -Wl,--stack,268435456)   # 256 MB main stack: the lifted code nests one C frame per MIPS call, and the GL driver runs on top of it
else()
  set(OpenGL_GL_PREFERENCE GLVND)
  find_package(OpenGL REQUIRED)
  set(GW_TARGETS gunwars gunwars_trace)
  set(GW_SYS_LIBS OpenGL::GL m)
  set(GW_LINK_OPTS "")
endif()
foreach(t ${GW_TARGETS})
  add_executable(${t} ${GW_SRC})
  # include/ first: its tc2_lifted.h names THIS game's lifted program
  target_include_directories(${t} PRIVATE include gen ${S23_HOST_INCLUDE} ${SDL2_INCLUDE_DIRS})
  if(WIN32)
    target_link_libraries(${t} ${SDL2_STATIC_LDFLAGS} ZLIB::ZLIB Threads::Threads ${GW_SYS_LIBS})
  else()
    target_link_libraries(${t} ${SDL2_LIBRARIES} ZLIB::ZLIB Threads::Threads ${GW_SYS_LIBS})
  endif()
  target_link_options(${t} PRIVATE ${GW_LINK_OPTS})
endforeach()
if(TARGET gunwars_trace)
  target_compile_definitions(gunwars_trace PRIVATE RR_TRACE)
endif()
